Create Customer

curl -X POST -H "Content-Type: application/json" \
  --data '{"query":"mutation { createCustomer(input: { name: \"john\", cpf: \"123.456.789-00\", email: \"john@gmail.com\" }) { id } }"}' \
  http://localhost:8080/graphql

Get Customer by ID

curl -X POST -H "Content-Type: application/json" \
  --data '{"query": "query { customerOfId(id: \"cbf4b763-dc0b-427d-ba31-bf593989c2f6\") { name } }"}' \
  http://localhost:8080/graphql
